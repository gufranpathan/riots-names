% Frame title: serif, with a short crimson rule beneath. The \strut gives the
% title's last line the same depth whether or not it has letters that descend
% (g, p, y), so the rule sits at the same height on every slide.
\setbeamertemplate{frametitle}{%
  \vspace*{2.4mm}%
  {\headingfont\fontsize{15}{18}\selectfont\color{ink}\insertframetitle\strut\par}%
  \nointerlineskip\vspace{0.2mm}%
  \noindent{\color{crimson}\rule{11.3mm}{0.35mm}}\par
  \nointerlineskip%
}

% Title page, section pages and the appendix divider: a crimson bar to the
% left of the title, centred vertically. Beamer adds a fill at the bottom of
% top-aligned frames, so the fill above counts twice. Inside the TikZ node,
% align=left fixes word spaces at the size of the font current when the node
% starts; \spaceskip=0pt restores spaces that follow the font. The node is as
% wide as the text less the indent and the bar, so no line is overfull.
\newcommand{\barredtitle}[1]{%
  \vspace*{\stretch{2}}\vspace*{2mm}%
  \noindent\hspace*{4.8mm}\begin{tikzpicture}
    \node[anchor=north west, inner sep=0pt, text width=\textwidth-9.8mm, align=left] (t)
      {\hyphenpenalty=10000 \spaceskip=0pt \xspaceskip=0pt #1};
    \fill[crimson] ([xshift=-4.8mm, yshift=0.6mm]t.north west)
      rectangle ([xshift=-4.1mm, yshift=-0.6mm]t.south west);
  \end{tikzpicture}%
  \vspace*{\fill}%
}
% The subtitle and author lines are printed only when the YAML sets them.
\makeatletter
\newcommand{\ifauthorset}[2]{\ifx\beamer@shortauthor\@empty#2\else#1\fi}
\newcommand{\ifsubtitleset}[2]{\ifx\insertsubtitle\@empty#2\else#1\fi}
\makeatother
\setbeamertemplate{title page}{%
  \barredtitle{%
    {\headingfont\fontsize{19.5}{23}\selectfont\color{ink}\inserttitle\par}%
    \ifsubtitleset{\vspace{2.2mm}%
      {\subheadingfont\fontsize{10.5}{13}\selectfont\color{muted}\insertsubtitle\par}}{}%
    \vspace{7.6mm}%
    {\fontsize{7.9}{10.5}\selectfont\color{muted}%
      \ifauthorset{\insertauthor\par\vspace{1.5mm}}{}\insertdate\par}%
  }%
}
\setbeamertemplate{section page}{%
  \barredtitle{{\headingfont\fontsize{19.5}{23}\selectfont\color{ink}\insertsectionhead\par}}%
}
% A level-1 heading starts a numbered section page (it replaces pandoc's
% \AtBeginSection, which comes earlier in the preamble).
\AtBeginSection{\begin{frame}\sectionpage\end{frame}}

% Appendix. beamer.lua turns the heading `# Appendix {#appendix}` into an
% unnumbered frame that calls \appendixdivider, and marks every later frame
% noframenumbering. The slide total then counts the main slides only, and the
% footer shows no number in the appendix.
\newif\ifinappendix
\newcommand{\appendixdivider}[1]{%
  \global\inappendixtrue
  \barredtitle{{\headingfont\fontsize{19.5}{23}\selectfont\color{ink}#1\par}}}

% Footer: slide number at the right, a thin crimson progress bar at the bottom
\setbeamertemplate{footline}{%
  % Right-aligned with the text's right margin (\hspace*: TeX drops a plain
  % \hspace at the end of a line), 1 mm below the text area
  \vspace*{1mm}\hfill{\fontsize{4.9}{6}\selectfont\color{muted}\ifinappendix\strut\else\insertframenumber\,/\,\inserttotalframenumber\fi}\hspace*{6.4mm}\par
  \vspace{1.1mm}%
  \noindent\begin{tikzpicture}
    \fill[rulegrey] (0,0) rectangle (\paperwidth,0.32mm);
    % min(): on LaTeX's first pass the total is 1, and from about the 37th
    % frame on, frame x width exceeds TeX's largest dimension
    \fill[crimson] (0,0) rectangle ({min(\insertframenumber/\inserttotalframenumber,1)*\paperwidth},0.32mm);
  \end{tikzpicture}%
}

% Pandoc marks short lists as tight: \tightlist sets \itemsep and \parskip to
% 0. Most of the space between items on a slide is pandoc's \parskip (6 pt),
% so \tightlist does nothing here and tight lists keep that space.
% (beamer.lua restores \parskip inside columns, which are minipages.)
\AtBeginDocument{\renewcommand{\tightlist}{}}
\setlength{\leftmargini}{1.4em}
\setlength{\leftmarginii}{1.3em}

\newcommand{\chip}[1]{% a link chip: outlined monospace label
  \tikz[baseline=(c.base)]\node[draw=rulegrey, line width=0.36pt, rounded corners=1.4pt,
    inner xsep=4.6pt, inner ysep=2.4pt, text=ink,
    font=\plainmono\fontsize{6.5}{7.5}\selectfont] (c) {\vphantom{Ap}#1};}
\newcommand{\backlink}[2]{% target, text: pinned to the bottom left of the frame
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=base west, inner sep=0pt] at ([xshift=6.4mm, yshift=4.3mm]current page.south west)
      {\backlinktext{#1}{#2}};
  \end{tikzpicture}}
\newcommand{\backlinktext}[2]{%
  \hyperlink{#1}{\tikz[baseline=(b.base)]{%
    \node[inner sep=0pt, text=muted, font=\fontsize{5.9}{7.5}\selectfont] (b) {#2};
    \draw[rulegrey, line width=0.36pt] ([yshift=-0.8pt]b.south west) -- ([yshift=-0.8pt]b.south east);}}}
